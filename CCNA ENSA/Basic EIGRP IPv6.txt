# Router 1
hostname R1
ipv6 unicast routing
interface g0/0/0
ipv6 address 2001:db8:acad:a::1/64
ipv6 address fe80::1 link-local
Exit
interface s0/2/0 
ipv6 address 2001:db8:acad:13::1/64
ipv6 address fe80::1 link-local
Exit
interface s0/2/1
ipv6 address 2001:db8:acad:13::1/64
ipv6 address fe80::1 link-local
Exit

ipv6 router eigrp 1
router id 1.1.1.1
no shut
Exit

interface g0/0/0
ipv6 router eigrp 1
Exit
interface s0/2/0
ipv6 router eigrp 1
Exit
interface s0/2/1
ipv6 router eigrp 1
Exit
